{{- /* Figure Shortcode - Hugo v0.157.0+ WebP optimized figure with Hugo Pipes Usage: {{< figure src="/images/photo.jpg" alt="Description" caption="Caption" width="800" >}} Parameters: - src: Image path (required, must start with /) - alt: Alt text (required) - caption: Figure caption (optional) - class: CSS class (optional) - width: Resize width (optional, default: 800) - quality: Image quality 1-100 (optional, default: 80) - loading: lazy/eager (optional, default: lazy) - link: Optional link URL (optional) */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- $class := .Get "class" | default "" -}} {{- $width := .Get "width" | default 800 -}} {{- $quality := .Get "quality" | default 80 -}} {{- $loading := .Get "loading" | default "lazy" -}} {{- $link := .Get "link" -}} {{- $img := resources.Get $src -}} {{- $imgHTML := "" -}} {{- if $img -}} {{- $resizeOp := printf "%dx webp q%d" $width $quality -}} {{- $processed := try ($img.Resize $resizeOp) -}} {{- if $processed.Err -}} {{- /* Image processing failed, render fallback */ -}} {{- $imgHTML = printf `` $src $alt $loading -}} {{- else -}} {{- $result := $processed.Value -}} {{- $imgHTML = printf `` $result.RelPermalink $alt $loading $result.Width $result.Height -}} {{- end -}} {{- end -}}
